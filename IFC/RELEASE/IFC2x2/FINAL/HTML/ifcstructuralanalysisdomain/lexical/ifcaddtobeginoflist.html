FUNCTION IfcAddToBeginOfList
  (AScalar : GENERIC;
   AList   : LIST OF GENERIC)
  :LIST[1:?] OF GENERIC;

  LOCAL
    Result : LIST OF GENERIC := [];
  END_LOCAL;

  IF NOT(EXISTS(AScalar)) THEN
    Result := AList;
  ELSE
    Result := Result + AScalar;
    IF (HIINDEX(AList) >= 1) THEN
      REPEAT i:=1 TO HIINDEX(AList);
        Result[i+1] := AList[i];
      END_REPEAT;
    END_IF;
  END_IF;
  RETURN(Result);
END_FUNCTION;
